system:
  name: Deterministic Agentic Control System
  short_name: DACS
  purpose: Control, improve, and build deterministic agents under measurable governance.
  default_autonomy_level: 3
  default_autonomy_name: Sandbox

principles:
  - Same input plus same state plus same policy version should produce the same output.
  - Agents may propose and test improvements, but production changes require approval.
  - Every agent must have explicit input schema, output schema, state model, rules, tests, metrics, and logs.
  - Agents cannot expand their own permissions.
  - Every change must be versioned and rollbackable.

planes:
  control_plane:
    components:
      - supervisor_agent
      - policy_engine
      - permission_gateway
      - agent_registry
      - change_control
  runtime_plane:
    components:
      - deterministic_worker_agents
      - workflow_engine
      - state_store
      - tool_gateway
  build_plane:
    components:
      - agent_factory
      - template_library
      - capability_mapper
      - spec_builder
      - test_builder
  evaluation_plane:
    components:
      - metrics_collector
      - trace_store
      - regression_runner
      - determinism_tester
      - policy_tester
  improvement_plane:
    components:
      - metrics_analyst_agent
      - root_cause_agent
      - patch_proposal_agent
      - review_gate_agent
  deployment_plane:
    stages:
      - draft
      - sandbox
      - evaluation
      - review
      - staging
      - production
      - monitoring

agents:
  supervisor_agent:
    role: Orchestrates the system and assigns work to specialized agents.
    may_deploy_to_production: false
    approval_required_for:
      - production_change
      - permission_expansion
      - destructive_action
  policy_agent:
    role: Enforces policies, permissions, and safety constraints.
    may_block_actions: true
  builder_agent:
    role: Creates new deterministic agent drafts from templates.
    allowed_environments:
      - draft
      - sandbox
    may_deploy_to_production: false
  evaluation_agent:
    role: Runs determinism, regression, policy, and performance tests.
    may_deploy_to_production: false
  improvement_agent:
    role: Finds gaps and proposes controlled improvements.
    may_change_production: false
  review_gate_agent:
    role: Checks whether a proposed build or improvement is safe to review.
    may_approve_final_production_deploy: false

agent_requirements:
  required_fields:
    - agent_id
    - name
    - purpose
    - owner
    - version
    - framework
    - input_schema
    - output_schema
    - state_model
    - decision_rules
    - tie_break_policy
    - allowed_tools
    - evaluation_suite
    - metrics
    - logs
    - rollback_version

frameworks:
  rule_based_system:
    best_for: Simple explicit business logic.
  decision_table:
    best_for: Many condition and action combinations.
  finite_state_machine:
    best_for: Strict lifecycle control.
  workflow_engine:
    best_for: Multi-step operational orchestration.
  constraint_based_system:
    best_for: Hard policy and safety boundaries.
  deterministic_scoring:
    best_for: Ranking and prioritization.
  hybrid_rule_workflow:
    best_for: Production operational automation.

improvement_loop:
  stages:
    - measure
    - compare
    - diagnose
    - propose
    - test
    - review
    - deploy
    - monitor
  production_deployment_requires:
    - passing_tests
    - policy_check
    - rollback_plan
    - approval

metrics:
  reliability:
    - success_rate
    - failure_rate
    - retry_rate
    - verification_failure_rate
  determinism:
    - same_input_same_output_rate
    - tie_break_usage_count
    - rule_conflict_count
  operations:
    - average_latency
    - throughput
    - escalation_rate
    - human_override_rate
  safety:
    - policy_violation_count
    - unauthorized_tool_attempt_count
  audit:
    - trace_completeness
    - policy_version_coverage
    - missing_log_rate

control_modes:
  - level: 0
    name: Manual
    production_changes_allowed: false
  - level: 1
    name: Recommend
    production_changes_allowed: false
  - level: 2
    name: Draft
    production_changes_allowed: false
  - level: 3
    name: Sandbox
    production_changes_allowed: false
  - level: 4
    name: Staged
    production_changes_allowed: approval_required
  - level: 5
    name: Production
    production_changes_allowed: explicit_approval_required

guardrails:
  - No silent self-modification of production agents.
  - No production deployment without approval.
  - No permission expansion without approval.
  - No unversioned rule changes.
  - No hidden state as a decision source.
  - No change without regression tests.
  - No external side effects from sandbox tests unless explicitly allowed.
